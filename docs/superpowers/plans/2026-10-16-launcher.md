# Native macOS Launcher Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A native Mac launcher window for `Bloodborne.app` (0.4) that sets display, upscaler, effects, frame rate cap and folders, then starts the game.

**Architecture:** A small AppKit + SwiftUI app (`launcher/`, built with `swiftc`, no Xcode project) becomes `Contents/MacOS/Bloodborne`. Pure model files (ini parsing, settings, resolutions, launch plan) are unit-tested by a separate test executable; the UI and process code sit on top. Three small engine changes make the settings reach the game: `BB_SAVE_DIR`, `fps_cap`/`fullscreen` in `run.sh`, and an in-game menu that keeps keys it does not own.

**Tech Stack:** Swift 6 (swiftc from Xcode), SwiftUI hosted in an AppKit `NSWindow`, bash (run.sh), C (runtime), C++20 (settings).

**Spec:** `docs/superpowers/specs/2026-10-16-launcher-design.md`

## Review Focus

1. **Paths with spaces or quotes** (`~/Library/Application Support/…`, `/Volumes/My Drive/…`,
   a folder named `Hunter's saves`): `launch.conf` must round-trip and stay valid for bash
   `source` (Bloodborne.command reads it). Test in Task 5 (Swift writes, bash sources).
2. **Values the launcher does not offer** in `bbport.ini` (`output_res=3456x1944` from the Metal
   branch, `upscaler=fsr4`): shown as they are and written back unchanged unless the user picks
   something else. Test in Task 5.
3. **The in-game menu changes settings while the launcher waits**: after the game exits the
   launcher reloads `bbport.ini` before showing it again, so it never writes stale values over
   the menu's. Test in Task 5 (reload) and by hand in Task 9.
4. **Save folder edge cases**: choosing the same folder, or a folder that already has saves, must
   not offer a copy; a copy never overwrites or deletes. Test in Task 5.
5. **The game ends at once** (run.sh error, game check failure, already running): the launcher
   comes back with the status and Show Log instead of hanging in "Running…". By hand in Task 9.

# SPDX-License-Identifier: GPL-2.0-or-later
# Sourced by run.sh: bbport.ini settings the launcher writes that become environment variables.
#   fps_cap=30|40|60|90|120  the presenter lets at most that many frames through
#                            (BB_FPS_LIMIT); the game keeps run.sh's default frame rate patch
#   fullscreen=1             the window opens full screen (BB_FULLSCREEN=1)
# Values already in the environment win. Bash 3.2 (macOS /bin/bash) compatible.
bb_launch_settings() { # ini file
    local line cap='' full=''
    [[ -f $1 ]] || return 0
    while IFS= read -r line || [[ -n $line ]]; do
        line=${line%$'\r'}
        if [[ $line =~ ^fps_cap=(30|40|60|90|120)$ ]]; then cap=${BASH_REMATCH[1]}; fi
        if [[ $line =~ ^fullscreen=([01])$ ]]; then full=${BASH_REMATCH[1]}; fi
    done < "$1"
    if [[ -n $cap && -z ${BB_FPS_LIMIT:-} ]]; then export BB_FPS_LIMIT=$cap; fi
    if [[ $full == 1 && -z ${BB_FULLSCREEN:-} ]]; then export BB_FULLSCREEN=1; fi
    return 0
}

// SPDX-License-Identifier: GPL-2.0-or-later
import Foundation

/// A `key=value` file (bbport.ini, launch.conf). Comments, blank lines, unknown keys and their
/// order survive a read and write; only the keys that are set change. Parsing follows the game's
/// own reader (gpu/shim/bbport_settings.cpp Load): the key is what precedes the first `=`.
struct IniFile: Equatable {
    enum Line: Equatable {
        case entry(key: String, value: String)
        case other(String)
    }

    private(set) var lines: [Line] = []

    init(text: String = "") {
        for raw in text.split(separator: "\n", omittingEmptySubsequences: false) {
            var line = String(raw)
            if line.hasSuffix("\r") {
                line.removeLast()
            }
            if !line.hasPrefix("#"), let eq = line.firstIndex(of: "=") {
                let key = line[..<eq].trimmingCharacters(in: .whitespaces)
                let value = line[line.index(after: eq)...].trimmingCharacters(in: .whitespaces)
                if !key.isEmpty {
                    lines.append(.entry(key: key, value: value))
                    continue
                }
            }
            lines.append(.other(line))
        }
        // The final newline leaves one empty piece: not a line of the file.
        if case .other("") = lines.last {
            lines.removeLast()
        }
    }

    init(contentsOf url: URL) {
        self.init(text: (try? String(contentsOf: url, encoding: .utf8)) ?? "")
    }

    private func isEntry(_ line: Line, _ key: String) -> Bool {
        if case let .entry(k, _) = line { return k == key }
        return false
    }

    subscript(key: String) -> String? {
        get {
            for line in lines.reversed() {
                if case let .entry(k, v) = line, k == key { return v }
            }
            return nil
        }
        set {
            guard let value = newValue else {
                lines.removeAll { isEntry($0, key) }
                return
            }
            if let i = lines.lastIndex(where: { isEntry($0, key) }) {
                lines[i] = .entry(key: key, value: value)
            } else {
                lines.append(.entry(key: key, value: value))
            }
        }
    }

    var text: String {
        lines.map { line -> String in
            switch line {
            case let .entry(key, value): return "\(key)=\(value)"
            case let .other(text): return text
            }
        }.joined(separator: "\n") + "\n"
    }

    func write(to url: URL) throws {
        try FileManager.default.createDirectory(at: url.deletingLastPathComponent(),
                                                withIntermediateDirectories: true)
        try text.write(to: url, atomically: true, encoding: .utf8)
    }
}

/// An output or render size in pixels.
struct PixelSize: Hashable, CustomStringConvertible {
    var width: Int
    var height: Int

    init(_ width: Int, _ height: Int) {
        self.width = width
        self.height = height
    }

    /// "2560x1440", as bbport.ini's output_res.
    init?(iniValue: String) {
        let parts = iniValue.lowercased().split(separator: "x")
        guard parts.count == 2, let w = Int(parts[0]), let h = Int(parts[1]), w > 0, h > 0 else {
            return nil
        }
        self.init(w, h)
    }

    var iniValue: String { "\(width)x\(height)" }
    var description: String { "\(width) × \(height)" }
}

enum Resolution {
    /// The in-game menu's output list (gpu/shim/bbport_settings.h OutputWidths / OutputHeights).
    static let outputs = [PixelSize(1280, 720), PixelSize(1920, 1080), PixelSize(2560, 1440),
                          PixelSize(3840, 2160)]
    static let defaultOutput = PixelSize(1920, 1080)
    /// Upscaler quality levels (bbport_settings.cpp PresetName / PresetScale).
    static let presetNames = ["Native AA", "Quality", "Balanced", "Performance", "Ultra Performance"]
    static let presetScales = [1.0, 1.5, 1.7, 2.0, 3.0]

    /// The scene size the game renders for an output and quality level: even sizes, rounded as
    /// scripts/patches.py does (its round() ties to even; no output in the list hits a tie).
    static func renderSize(output: PixelSize, preset: Int) -> PixelSize {
        let scale = presetScales[min(max(preset, 0), presetScales.count - 1)]
        func side(_ v: Int) -> Int { max(2, Int((Double(v) / scale / 2).rounded()) * 2) }

// SPDX-License-Identifier: GPL-2.0-or-later
import Foundation

enum Shell {
    /// Single-quoted for bash: it's -> 'it'\''s'.
    static func quote(_ s: String) -> String {
        "'" + s.replacingOccurrences(of: "'", with: "'\\''") + "'"
    }

    /// The value of a shell word: single quotes, double quotes and backslash escapes (what
    /// quote() and the earlier launchers' printf %q wrote).
    static func unquote(_ s: String) -> String {
        var out = ""
        var single = false, double = false, escaped = false
        for c in s {
            if escaped {
                out.append(c)
                escaped = false
            } else if single {
                if c == "'" { single = false } else { out.append(c) }
            } else if c == "\\" {
                escaped = true
            } else if c == "'" && !double {
                single = true
            } else if c == "\"" {
                double.toggle()
            } else {
                out.append(c)
            }
        }
        return out
    }

/// launch.conf: the launcher's own settings as shell assignments (bash sources the file in

struct LaunchConfig: Equatable {

    var saveDir: URL?
    var showLauncher = true

        dataDir = path("BB_DATA_DIR")

        ini["BB_GAME_DIR"] = gameDir.map { Shell.quote($0.path) }
        ini["BB_SAVE_DIR"] = saveDir.map { Shell.quote($0.path) }

import Foundation

/// The save folder: BB_SAVE_DIR for the game (src/runtime_savedata.c), default <data>/user/savedata.
enum SaveFolder {
    static func defaultURL(dataDir: URL) -> URL {
        dataDir.appendingPathComponent("user/savedata")
    }

    /// Any regular file below the folder.
    static func hasSaves(_ url: URL) -> Bool {
        guard let walk = FileManager.default.enumerator(at: url, includingPropertiesForKeys: [.isRegularFileKey]) else {
            return false
        }
        for case let file as URL in walk {
            if (try? file.resourceValues(forKeys: [.isRegularFileKey]).isRegularFile) == true { return true }
        }
        return false
    }

    /// Moving from a folder with saves to another without any: offer to copy them.
    static func shouldOfferCopy(from old: URL, to new: URL) -> Bool {
        old.standardizedFileURL.path != new.standardizedFileURL.path && hasSaves(old) && !hasSaves(new)
    }

    /// Copies the save tree; never overwrites a file in `new`, never touches `old`.
    static func copySaves(from old: URL, to new: URL) throws {
        let fm = FileManager.default
        try fm.createDirectory(at: new, withIntermediateDirectories: true)
        for item in try fm.contentsOfDirectory(at: old, includingPropertiesForKeys: [.isDirectoryKey]) {
            let target = new.appendingPathComponent(item.lastPathComponent)
            let isDirectory = (try? item.resourceValues(forKeys: [.isDirectoryKey]).isDirectory) == true
            if isDirectory && fm.fileExists(atPath: target.path) {
                try copySaves(from: item, to: target)
            } else if !fm.fileExists(atPath: target.path) {
                try fm.copyItem(at: item, to: target)
            }
        }
    }

    /// nil when the game can write saves there, else why not. `create`: make the folder (when the
    /// player chose it); otherwise a missing folder (an unplugged drive) is a problem.
    static func problem(_ url: URL, create: Bool) -> String? {
        let fm = FileManager.default
        var isDirectory: ObjCBool = false
        if !fm.fileExists(atPath: url.path, isDirectory: &isDirectory) {
            guard create else { return "The save folder \(url.path) is missing (is its drive connected?)." }
            do {
                try fm.createDirectory(at: url, withIntermediateDirectories: true)
            } catch {
                return "Cannot create \(url.path): \(error.localizedDescription)"
            }
        } else if !isDirectory.boolValue {
            return "\(url.path) is a file, not a folder."
        }
        return fm.isWritableFile(atPath: url.path) ? nil : "\(url.path) is not writable."
    }
}

import Foundation

/// How the launcher starts the game: the program, arguments, environment and folder.
struct LaunchPlan: Equatable {
    var executable: URL
    var arguments: [String]
    var environment: [String: String]
    var directory: URL
}

    /// game.log (this session) and game.previous.log.
    static func folder(home: URL) -> URL { home.appendingPathComponent("Library/Logs/Bloodborne") }

/// Where the parts are: the release app (everything inside it, scripts/macos/package_app.sh) or
/// the checkout's local app (scripts/macos/make_app.sh; Info.plist BBRepository = the checkout).
enum Layout: Equatable {
    case packaged(contents: URL)
    case checkout(repo: URL, app: URL)

    static func detect(bundleURL: URL, info: [String: Any]) -> Layout? {
        let contents = bundleURL.appendingPathComponent("Contents")
        if FileManager.default.fileExists(atPath: contents.appendingPathComponent("Resources/port/run.sh").path) {
            return .packaged(contents: contents)
        }
        if let repo = info["BBRepository"] as? String, !repo.isEmpty {
            return .checkout(repo: URL(fileURLWithPath: repo), app: bundleURL)
        }
        return nil
    }

    private var port: URL? {
        if case let .packaged(contents) = self { return contents.appendingPathComponent("Resources/port") }
        return nil
    }

    var hasMetalRenderer: Bool {
        switch self {
        case let .packaged(contents):
            FileManager.default.fileExists(atPath: contents.appendingPathComponent("Resources/port/metal-renderer").path)
        case let .checkout(repo, _):
            FileManager.default.fileExists(atPath: repo.appendingPathComponent("gpu/shadps4/video_core/renderer_metal").path)
        }
    }

    private static func home(_ env: [String: String]) -> URL {
        URL(fileURLWithPath: env["HOME"] ?? NSHomeDirectory())
    }

    private static func defaultData(_ env: [String: String]) -> URL {
        home(env).appendingPathComponent("Library/Application Support/Bloodborne")
    }

    /// launch.conf: in the data folder for the release app, next to Bloodborne.command in a
    /// checkout (shared with it). BB_LAUNCH_CONF overrides (tests).
    func launchConfURL(env: [String: String]) -> URL {
        if let path = env["BB_LAUNCH_CONF"], !path.isEmpty { return URL(fileURLWithPath: path) }
        switch self {
        case .packaged: return Self.defaultData(env).appendingPathComponent("launch.conf")
        case let .checkout(repo, _): return repo.appendingPathComponent("launch.conf")
        }
    }

    /// bbport.ini, the shader cache and the default save folder: BB_DATA_DIR, else launch.conf's
    /// BB_DATA_DIR, else ~/Library/Application Support/Bloodborne (run.sh's default).
    func dataDir(env: [String: String], config: LaunchConfig) -> URL {
        if let path = env["BB_DATA_DIR"], !path.isEmpty { return URL(fileURLWithPath: path) }
        return config.dataDir ?? Self.defaultData(env)
    }

    /// The game folder to suggest: ../CUSA03173 next to a checkout (as Bloodborne.command does).
    func defaultGameDir() -> URL? {
        guard case let .checkout(repo, _) = self else { return nil }
        let game = repo.deletingLastPathComponent().appendingPathComponent("CUSA03173")
        return FileManager.default.fileExists(atPath: game.appendingPathComponent("eboot.bin").path) ? game : nil
    }

    /// scripts/game_check.py with the Python run.sh uses (prints why a folder cannot be used).
    var gameCheckCommand: [String] {
        switch self {
        case let .packaged(contents):
            [contents.appendingPathComponent("Resources/python/bin/python3").path,
             contents.appendingPathComponent("Resources/port/scripts/game_check.py").path]
        case let .checkout(repo, _):
            ["/usr/bin/env", "python3", repo.appendingPathComponent("scripts/game_check.py").path]
        }
    }

    /// The environment scripts/macos/app_launcher.sh set (release app) or Bloodborne.app's
    /// earlier launcher (checkout), plus the save folder. Display settings reach the game through
    /// bbport.ini (run.sh: scripts/launch_settings.sh), not here.
    func launchPlan(gameDir: URL, saveDir: URL?, dataDir: URL, env: [String: String]) -> LaunchPlan {
        var environment = env
        environment["BB_GAME_DIR"] = gameDir.path
        if let saveDir {
            environment["BB_SAVE_DIR"] = saveDir.path
        } else {
            environment.removeValue(forKey: "BB_SAVE_DIR")
        }
        let bash = URL(fileURLWithPath: "/bin/bash")
        switch self {
        case let .packaged(contents):
            let port = self.port!
            environment["BB_DATA_DIR"] = dataDir.path
            environment["BB_PREBUILT"] = "1"
            environment["BB_PROBE"] = contents.appendingPathComponent("MacOS/bb-probe").path
            environment["PYTHON"] = contents.appendingPathComponent("Resources/python/bin/python3").path
            environment["PYTHONDONTWRITEBYTECODE"] = "1"
            environment["VK_DRIVER_FILES"] = port.appendingPathComponent("MoltenVK_icd.json").path
            return LaunchPlan(executable: bash, arguments: [port.appendingPathComponent("run.sh").path],
                              environment: environment, directory: port)
        case let .checkout(repo, app):
            // run.sh rebuilds a checkout: Finder's PATH lacks Homebrew's cmake and ninja.
            environment["PATH"] = "/opt/homebrew/bin:" + (env["PATH"] ?? "/usr/bin:/bin:/usr/sbin:/sbin")
            environment["BB_GPU_HOST"] = app.appendingPathComponent("Contents/MacOS/bbgpu-host").path
            return LaunchPlan(executable: bash, arguments: [repo.appendingPathComponent("run.sh").path],
                              environment: environment, directory: repo)
        }
    }

// SPDX-License-Identifier: GPL-2.0-or-later
import AppKit
import Foundation

/// Process work for the launcher: the game, the game check, Rosetta.
enum GameRunner {
    /// Runs a short command; its status and output (stdout and stderr together).
    @discardableResult
    static func run(_ command: [String]) -> (status: Int32, output: String) {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: command[0])
        process.arguments = Array(command.dropFirst())
        let pipe = Pipe()
        process.standardOutput = pipe
        process.standardError = pipe
        do { try process.run() } catch { return (-1, error.localizedDescription) }
        let data = pipe.fileHandleForReading.readDataToEndOfFile()
        process.waitUntilExit()
        return (process.terminationStatus, String(decoding: data, as: UTF8.self))
    }

    /// The game's processes (one copy at a time).
    static var isGameRunning: Bool {
        run(["/usr/bin/pgrep", "-x", "bbgpu-host"]).status == 0 || run(["/usr/bin/pgrep", "-x", "bb-probe"]).status == 0
    }

    static var hasRosetta: Bool { run(["/usr/bin/arch", "-x86_64", "/usr/bin/true"]).status == 0 }

    /// Asks for an administrator password (macOS's dialog) and installs Rosetta 2.
    static func installRosetta() -> Bool {
        run(["/usr/bin/osascript", "-e",
             "do shell script \"/usr/sbin/softwareupdate --install-rosetta --agree-to-license\" with administrator privileges"])
        return hasRosetta
    }

    /// nil when the folder holds a usable game, else game_check.py's reason.
    static func checkGame(_ folder: URL, layout: Layout) -> String? {
        guard FileManager.default.fileExists(atPath: folder.appendingPathComponent("eboot.bin").path) else {
            return "There is no eboot.bin in this folder. Choose the folder that contains eboot.bin, sce_sys and dvdroot_ps4."
        }
        let result = run(layout.gameCheckCommand + [folder.path])
        if result.status == 0 { return nil }
        let reason = result.output.trimmingCharacters(in: .whitespacesAndNewlines)
        return reason.isEmpty ? "The game check failed (status \(result.status))." : reason
    }

    /// Starts the game; `done` gets its exit status on the main queue. The output goes to
    /// `log` (the previous session's log is kept as game.previous.log).
    static func start(_ plan: LaunchPlan, log: URL, done: @escaping (Int32) -> Void) throws -> Process {
        let fm = FileManager.default
        try fm.createDirectory(at: log.deletingLastPathComponent(), withIntermediateDirectories: true)
        let previous = log.deletingLastPathComponent().appendingPathComponent("game.previous.log")
        if fm.fileExists(atPath: log.path) {
            try? fm.removeItem(at: previous)
            try? fm.moveItem(at: log, to: previous)
        }
        fm.createFile(atPath: log.path, contents: nil)
        let output = try FileHandle(forWritingTo: log)
        let process = Process()
        process.executableURL = plan.executable
        process.arguments = plan.arguments
        process.environment = plan.environment
        process.currentDirectoryURL = plan.directory
        process.standardInput = FileHandle.nullDevice
        process.standardOutput = output
        process.standardError = output
        process.terminationHandler = { finished in
            try? output.close()
            DispatchQueue.main.async { done(finished.terminationStatus) }
        }
        try process.run()
        return process
    }
}

        panel.directoryURL = start
        return panel.runModal() == .OK ? panel.url : nil
    }

    func chooseGameFolder() {
        guard let url = pickFolder("Choose your Bloodborne game folder (CUSA03173, the one containing eboot.bin).",
                                   start: config.gameDir) else { return }
        config.gameDir = url
        checkGame()
    }

    func resetGameFolder() {
        config.gameDir = layout.defaultGameDir()
        checkGame()
    }

    func chooseSaveFolder() {
        guard let url = pickFolder("Choose the folder for your saves.", start: saveDir) else { return }
        setSaveFolder(url)
    }

    func resetSaveFolder() { setSaveFolder(SaveFolder.defaultURL(dataDir: dataDir)) }

    private func setSaveFolder(_ url: URL) {
        let old = saveDir
        if let problem = SaveFolder.problem(url, create: true) {
            saveProblem = problem
            return
        }
        if SaveFolder.shouldOfferCopy(from: old, to: url) {
            let alert = NSAlert()
            alert.messageText = "Copy your saves to the new folder?"
            alert.informativeText = "Your saves in \(old.path) are copied to \(url.path). The originals stay where they are."
            alert.addButton(withTitle: "Copy Saves")
            alert.addButton(withTitle: "Don't Copy")
            if alert.runModal() == .alertFirstButtonReturn {
                do {
                    try SaveFolder.copySaves(from: old, to: url)
                } catch {
                    saveProblem = "Copying the saves failed: \(error.localizedDescription)"
                    return
                }
            }
        }
        config.saveDir = url.standardizedFileURL.path == SaveFolder.defaultURL(dataDir: dataDir).standardizedFileURL.path ? nil : url
        saveProblem = nil
    }

    func play() {
        guard !running, let game = config.gameDir else { return }
        save()
        if GameRunner.isGameRunning {
            exitMessage = "Bloodborne is already running. Only one copy of the game can run at a time."
            onExit?(-1)
            return
        }
        if !GameRunner.hasRosetta {
            let alert = NSAlert()
            alert.messageText = "Bloodborne needs Rosetta 2"
            alert.informativeText = "The game runs its original PlayStation 4 code through Rosetta 2, Apple's translator for Intel code. Install it now? (macOS asks for an administrator password.)"
            alert.addButton(withTitle: "Install")
            alert.addButton(withTitle: "Cancel")
            guard alert.runModal() == .alertFirstButtonReturn else { onExit?(-1); return }
            guard GameRunner.installRosetta() else {
                exitMessage = "Rosetta 2 was not installed. In Terminal, run: softwareupdate --install-rosetta"
                onExit?(-1)
                return
            }
        }
        if let custom = config.saveDir, let problem = SaveFolder.problem(custom, create: false) {
            let alert = NSAlert()
            alert.messageText = "The save folder is not available"
            alert.informativeText = problem
            alert.addButton(withTitle: "Use Default")
            alert.addButton(withTitle: "Cancel")
            guard alert.runModal() == .alertFirstButtonReturn else { onExit?(-1); return }
            config.saveDir = nil
            save()
        }
        let plan = layout.launchPlan(gameDir: game, saveDir: config.saveDir, dataDir: dataDir, env: env)
        do {
            _ = try GameRunner.start(plan, log: logURL) { [weak self] status in self?.finished(status) }
            running = true
            exitMessage = nil
            onStarted?()
        } catch {
            exitMessage = "The game could not start: \(error.localizedDescription)"
            onExit?(-1)
        }
    }

    private func finished(_ status: Int32) {
        running = false
        load()
        exitMessage = status == 0 ? nil : "Bloodborne stopped (status \(status))."
        onExit?(status)
    }

    func showLog() {
        GameRunner.run(["/usr/bin/open", "-a", "Console", logURL.path])
    }

    func effect(_ key: String) -> Bool { settings.effects[key] ?? false }
    func setEffect(_ key: String, _ on: Bool) { settings.effects[key] = on }
}

// Bloodborne.app's main executable: the launcher window (docs/superpowers/specs/2026-10-16-launcher-design.md).
import AppKit
import SwiftUI

@MainActor
final class AppDelegate: NSObject, NSApplicationDelegate {

    private var direct = false

    func applicationDidFinishLaunching(_ notification: Notification) {
        guard let layout = Layout.detect(bundleURL: Bundle.main.bundleURL,
                                         info: Bundle.main.infoDictionary ?? [:]) else {
            let alert = NSAlert()
            alert.messageText = "This copy of Bloodborne.app is incomplete"
            alert.informativeText = "Download the app again, or run scripts/macos/make_app.sh in the repository."
            alert.runModal()
            NSApp.terminate(nil)
            return
        }
        model = LauncherModel(layout: layout)
        model.onStarted = { [weak self] in
            self?.window?.orderOut(nil)
            NSApp.setActivationPolicy(.accessory) // the game's window has the Dock icon
        }
        model.onExit = { [weak self] status in self?.gameEnded(status) }
        model.load()
        if !NSEvent.modifierFlags.contains(.option) && model.canSkipWindow {
            direct = true
            model.play()
        } else {
            showWindow()
        }
    }

    private func gameEnded(_ status: Int32) {
        if direct && status == 0 {
            NSApp.terminate(nil)
            return
        }
        direct = false
        showWindow()
    }

    private func showWindow() {
        NSApp.setActivationPolicy(.regular)
        if window == nil {
            let window = NSWindow(contentRect: NSRect(x: 0, y: 0, width: 600, height: 760),
                                  styleMask: [.titled, .closable, .miniaturizable], backing: .buffered, defer: false)
            window.title = "Bloodborne"
            window.contentView = NSHostingView(rootView: LauncherView(model: model))
            window.isReleasedWhenClosed = false
            window.center()
            self.window = window
        }
        window?.makeKeyAndOrderFront(nil)
        NSApp.activate()
    }

    func applicationShouldTerminateAfterLastWindowClosed(_ sender: NSApplication) -> Bool {
        !model.running
    }

    func applicationWillTerminate(_ notification: Notification) {
        if model != nil && !model.running { model.save() }
    }
}
